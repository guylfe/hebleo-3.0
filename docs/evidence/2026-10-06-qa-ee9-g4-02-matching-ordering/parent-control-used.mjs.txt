import {readFileSync,writeFileSync} from 'node:fs';
import {spawnSync} from 'node:child_process';
import {createHash} from 'node:crypto';
const path='src/store.js',fixed=readFileSync(path),hash=b=>createHash('sha256').update(b).digest('hex');
const parent=spawnSync('git',['show','b312cdd294ace379080235d7b7275d99ca9e8612:src/store.js']);
if(parent.status!==0)throw new Error(parent.stderr.toString());
try {
  writeFileSync(path,parent.stdout);
  for(const [log,args] of [
    ['parent-store-focused.txt',['--test','test/matching-progress.test.js','test/ordering-tokens.test.js']],
    ['parent-store-app.txt',['--test','--test-name-pattern=QA-EE9-G4-02','test/app.test.js']]
  ]){
    const r=spawnSync(process.execPath,['../g402-run.mjs','docs/evidence/2026-10-06-qa-ee9-g4-02-matching-ordering/'+log,process.execPath,...args],{encoding:'utf8'});
    console.log(r.stdout,r.stderr,'expected pre-fix EXIT='+r.status);
  }
} finally {
  writeFileSync(path,fixed);
  if(hash(readFileSync(path))!==hash(fixed))throw new Error('Fixed store bytes not restored');
  console.log('Fixed store restored byte-for-byte SHA256='+hash(fixed));
}
