import {copyFileSync,mkdirSync,writeFileSync,existsSync} from 'node:fs';
import {fileURLToPath} from 'node:url';
import {randomUUID} from 'node:crypto';
import {Store} from '../src/store.js';
import {createServer} from '../src/server.js';
import {databaseReceipt} from '../scripts/runtime-state.mjs';
const dir=new URL('./classic-corrections-native/',import.meta.url);mkdirSync(dir,{recursive:true});
const path=new URL('candidate.sqlite',dir);if(existsSync(path))throw Error('Use a fresh disposable directory; never reuse an edited review database.');
const baseline=new URL('../../classic-editor-before-20261101/candidate.sqlite',import.meta.url),receipt=databaseReceipt(fileURLToPath(baseline));copyFileSync(baseline,path);
const store=new Store(fileURLToPath(path)),draft=store.currentDraft(),c=draft.content,lesson=c.lessons.find(l=>l.id==='MEMORY-A2'),block=lesson.blocks.find(b=>b.textPages),page=block.textPages.pages.find(p=>p.id===block.textPages.order[0]);
const source='## SYNTHETIC projection fixture\n\nSYNTHETIC repeated שָׁלוֹם 😀 quote.\n\nSYNTHETIC repeated שָׁלוֹם 😀 quote.\n';
page.html=source;page.title='SYNTHETIC projection verification';
store.stage({baseRevisionId:draft.baseRevisionId,expectedDraftToken:draft.draftToken,content:c});
const saved=store.currentDraft(),target={lessonId:lesson.id,baseRevisionId:saved.baseRevisionId,draftToken:saved.draftToken,componentId:block.id,pageId:page.id,activityId:null,fieldPath:'html'};
store.authorReview.create('author-demo',{...target,requestId:randomUUID(),selection:null,body:'SYNTHETIC preserved whole-block anchor'});
writeFileSync(new URL('PREPARED.json',dir),JSON.stringify({neverInstall:true,disposable:true,baselineReceipt:receipt.sha256,source,target,prepared:databaseReceipt(fileURLToPath(path)).sha256,pid:process.pid},null,2));
const server=createServer(store,{mediaRoot:fileURLToPath(new URL('media/',dir))});
server.listen(63483,'127.0.0.1',()=>console.log('DISPOSABLE correction review http://127.0.0.1:63483 PID '+process.pid));
process.on('SIGINT',()=>server.close(()=>{store.close();process.exit(0)}));
